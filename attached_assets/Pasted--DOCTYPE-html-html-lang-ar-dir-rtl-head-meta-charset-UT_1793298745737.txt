    <script>
        let currentRole = 'student';
        let currentLectureIndex = 0;

        let appState = JSON.parse(localStorage.getItem('uni_platform_data')) || {
            courseTitle: "تفاضل وتكامل ومفاهيم رياضية",
            instructorName: "أ.د. محمد أحمد",
            adminPassword: "416c6b75667261696e",
            serverApiUrl: "",
            lectures: [
                {
                    title: "المحاضرة 1: مقدمة في النظم والمتغيرات",
                    url: "https://commondatastorage.googleapis.com/gtv-videos-bucket/sample/ForBiggerBlazes.mp4",
                    desc: "نظرة عامة على متطلبات المادة وكيفية تحصيل الحضور والساعات غير المتزامنة.",
                    questions: [
                        {
                            q: "ما هو الشرط الأساسي لاحتساب الحضور في هذه المنصة؟",
                            options: ["مشاهدة الفيديو فقط", "حل الكويز وتسليم الاسم", "فتح الرابط وإغلاقه", "لا يوجد شرط"],
                            correct: 1
                        }
                    ]
                }
            ],
            attendanceLogs: []
        };

        function saveData() {
            localStorage.setItem('uni_platform_data', JSON.stringify(appState));
        }

        function initApp() {
            document.getElementById('nav-course-title').innerText = appState.courseTitle;
            document.getElementById('nav-instructor').innerText = appState.instructorName;
            document.getElementById('admin-course-input').value = appState.courseTitle;
            document.getElementById('admin-instructor-input').value = appState.instructorName;
            document.getElementById('admin-pass-input').value = appState.adminPassword || "416c6b75667261696e";
            document.getElementById('server-api-input').value = appState.serverApiUrl || "";
            renderLecturesList();
            loadLecture(currentLectureIndex);
            renderAttendanceTable();
        }

        function toggleView() {
            if (currentRole === 'student') {
                const currentPass = appState.adminPassword || "416c6b75667261696e";
                const pin = prompt("أدخل كلمة مرور الأدمن:");
                if (pin === currentPass) {
                    currentRole = 'admin';
                    document.getElementById('student-view').classList.add('hidden');
                    document.getElementById('admin-view').classList.remove('hidden');
                    document.getElementById('view-toggle-btn-text').innerText = "واجهة الطالب";
                    document.getElementById('role-badge').innerHTML = '<i class="fa-solid fa-user-shield ml-1"></i> لوحة الأدمن';
                } else if(pin !== null) {
                    alert("كلمة المرور خاطئة!");
                }
            } else {
                currentRole = 'student';
                document.getElementById('admin-view').classList.add('hidden');
                document.getElementById('student-view').classList.remove('hidden');
                document.getElementById('view-toggle-btn-text').innerText = "لوحة الأدمن";
                document.getElementById('role-badge').innerHTML = '<i class="fa-solid fa-user-graduate ml-1"></i> واجهة الطالب';
            }
        }

        function renderLecturesList() {
            const listEl = document.getElementById('lectures-list');
            listEl.innerHTML = '';
            appState.lectures.forEach((lec, idx) => {
                const item = document.createElement('div');
                item.className = `p-3.5 rounded-xl border transition-all cursor-pointer flex items-center justify-between ${idx === currentLectureIndex ? 'bg-purple-900/40 border-purple-500 text-white shadow-md' : 'bg-purple-950/20 border-purple-900/30 text-gray-400 hover:border-purple-700'}`;
                item.onclick = () => loadLecture(idx);
                item.innerHTML = `
                    <div class="flex items-center gap-3 truncate">
                        <i class="fa-regular fa-circle-play text-purple-400"></i>
                        <span class="text-sm font-semibold truncate">${lec.title}</span>
                    </div>
                `;
                listEl.appendChild(item);
            });
        }

        function loadLecture(index) {
            currentLectureIndex = index;
            const lec = appState.lectures[index];
            if(!lec) return;

            document.getElementById('lecture-code').innerText = `Lec ${index + 1}`;
            document.getElementById('lecture-title').innerText = lec.title;
            document.getElementById('lecture-desc').innerText = lec.desc;
            
            const video = document.getElementById('main-video');
            document.getElementById('video-source').src = lec.url;
            video.load();

            renderQuiz(lec.questions);
            renderLecturesList();
        }

        function renderQuiz(questions) {
            const container = document.getElementById('quiz-questions-container');
            container.innerHTML = '';
            if(!questions || questions.length === 0) {
                container.innerHTML = '<p class="text-xs text-gray-400">لا يوجد كويز مضاف لهذه المحاضرة.</p>';
                return;
            }
            questions.forEach((qObj, qIdx) => {
                const qDiv = document.createElement('div');
                qDiv.className = 'space-y-3';
                let optionsHtml = '';
                qObj.options.forEach((opt, oIdx) => {
                    optionsHtml += `
                        <label class="flex items-center gap-3 p-3 rounded-lg bg-slate-900/80 border border-purple-900/30 hover:border-purple-600 cursor-pointer text-sm text-gray-200">
                            <input type="radio" name="q_${qIdx}" value="${oIdx}" required class="accent-purple-500">
                            <span>${opt}</span>
                        </label>
                    `;
                });
                qDiv.innerHTML = `
                    <p class="font-bold text-sm text-purple-200">${qIdx + 1}. ${qObj.q}</p>
                    <div class="space-y-2">${optionsHtml}</div>
                `;
                container.appendChild(qDiv);
            });
        }

        function handleQuizSubmit(e) {
            e.preventDefault();
            const studentName = document.getElementById('student-name-input').value.trim();
            if(!studentName) {
                alert("يرجى إدخال اسمك الكامل أولاً لتسجيل الحضور!");
                return;
            }

            const currentLec = appState.lectures[currentLectureIndex];
            let score = 0;
            currentLec.questions.forEach((qObj, qIdx) => {
                const selected = document.querySelector(`input[name="q_${qIdx}"]:checked`);
                if(selected && parseInt(selected.value) === qObj.correct) {
                    score++;
                }
            });

            const totalQ = currentLec.questions.length;
            const logEntry = {
                studentName: studentName,
                lectureTitle: currentLec.title,
                score: `${score}/${totalQ}`,
                timestamp: new Date().toLocaleString('ar-JO')
            };

            appState.attendanceLogs.unshift(logEntry);
            saveData();
            renderAttendanceTable();

            // إذا تم ربط سيرفر، يمكن إرسال البيانات هنا عبر fetch(appState.serverApiUrl, { method: 'POST', body: JSON.stringify(logEntry) })

            document.getElementById('attendance-status').className = 'text-xs px-3 py-1.5 rounded-lg bg-emerald-950 text-emerald-300 font-semibold border border-emerald-800';
            document.getElementById('attendance-status').innerText = 'تم الحضور بنجاح';

            alert(`تم تسجيل حضورك بنجاح يا ${studentName}!\nعلامة الكويز: ${score}/${totalQ}`);
        }

        function saveCourseSettings() {
            appState.courseTitle = document.getElementById('admin-course-input').value;
            appState.instructorName = document.getElementById('admin-instructor-input').value;
            saveData();
            initApp();
            alert("تم حفظ إعدادات المادة والدكتور بنجاح!");
        }

        function saveAdminSecuritySettings() {
            const newPass = document.getElementById('admin-pass-input').value.trim();
            const newServerUrl = document.getElementById('server-api-input').value.trim();
            if(newPass) {
                appState.adminPassword = newPass;
            }
            appState.serverApiUrl = newServerUrl;
            saveData();
            alert("تم تحديث كلمة مرور الأدمن وإعدادات السيرفر بنجاح!");
        }

        function handleAddLecture(e) {
            e.preventDefault();
            const title = document.getElementById('new-lec-title').value;
            const url = document.getElementById('new-lec-url').value;
            const desc = document.getElementById('new-lec-desc').value;
            
            const qText = document.getElementById('new-q-text').value;
            const opts = [
                document.getElementById('new-opt-0').value,
                document.getElementById('new-opt-1').value,
                document.getElementById('new-opt-2').value,
                document.getElementById('new-opt-3').value
            ];
            const correctOpt = parseInt(document.getElementById('new-correct-opt').value);

            appState.lectures.push({
                title: title,
                url: url,
                desc: desc,
                questions: [{ q: qText, options: opts, correct: correctOpt }]
            });

            saveData();
            initApp();
            alert("تم إدراج المحاضرة والكويز بنجاح!");
        }

        function renderAttendanceTable() {
            const tbody = document.getElementById('attendance-table-body');
            tbody.innerHTML = '';
            if(appState.attendanceLogs.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" class="text-center p-4 text-xs text-gray-500">لا يوجد سجلات حضور حتى الآن.</td></tr>';
                return;
            }

            appState.attendanceLogs.forEach(log => {
                const tr = document.createElement('tr');
                tr.className = 'hover:bg-purple-950/30';
                tr.innerHTML = `
                    <td class="p-3 font-semibold text-white">${log.studentName}</td>
                    <td class="p-3 text-purple-300">${log.lectureTitle}</td>
                    <td class="p-3 font-mono text-purple-400 font-bold">${log.score}</td>
                    <td class="p-3 text-xs text-gray-400">${log.timestamp}</td>
                    <td class="p-3"><span class="px-2 py-1 rounded bg-emerald-950 text-emerald-400 text-xs border border-emerald-800">مكتمل</span></td>
                `;
                tbody.appendChild(tr);
            });
        }

        function exportAttendanceCSV() {
            if(appState.attendanceLogs.length === 0) {
                alert("لا توجد سجلات لتصديرها.");
                return;
            }
            let csvContent = "data:text/csv;charset=utf-8,\uFEFFاسم الطالب,المحاضرة,العلامة,التاريخ\n";
            appState.attendanceLogs.forEach(log => {
                csvContent += `"${log.studentName}","${log.lectureTitle}","${log.score}","${log.timestamp}"\n`;
            });
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "attendance_records.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        function importAttendanceFile(event) {
            const file = event.target.files[0];
            if(!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const content = e.target.result;
                    // دعم قراءة ملفات JSON كإستراد سجلات
                    const importedLogs = JSON.parse(content);
                    if(Array.isArray(importedLogs)) {
                        appState.attendanceLogs = importedLogs.concat(appState.attendanceLogs);
                        saveData();
                        renderAttendanceTable();
                        alert("تم إرفاق واستيراد السجلات بنجاح!");
                    } else {
                        alert("تنسيق الملف غير صحيح.");
                    }
                } catch(err) {
                    alert("حدث خطأ أثناء قراءة الملف المرفق. يجدر أن يكون الملف بصيغة JSON أو CSV متوافق.");
                }
            };
            reader.readAsText(file);
        }

        function clearRecords() {
            if(confirm("هل أنت تأكد من مسح كافة سجلات الحضور؟")) {
                appState.attendanceLogs = [];
                saveData();
                renderAttendanceTable();
            }
        }

        window.onload = initApp;
    </script>
